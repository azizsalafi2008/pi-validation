// Authenticate live user Pi.authenticate(['payments', 'username', 'wallet_address'], onIncompletePaymentFound) .then(auth => { currentUser = auth.user; statusBox.innerText = `Connected as: @${currentUser.username}`; payoutBtn.disabled = false; }) .catch(err => { statusBox.innerText = "Auth Error: " + err.message; }); function onIncompletePaymentFound(payment) { return fetch('https://www.mydocassistant.online/api/complete', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ paymentId: payment.identifier, txid: payment.transaction ? payment.transaction.txid : "incomplete_recovered" }) }); } // Step 1: User-to-App Payment function handlePayment() { payBtn.disabled = true; statusBox.innerText = "Opening payment window..."; Pi.createPayment({ amount: 0.1, memo: "DocHelper App Registration", metadata: { paymentType: "registration" } }, { onReadyForServerApproval: function(paymentId) { return fetch('https://www.mydocassistant.online/api/approve', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ paymentId: paymentId }) }); }, onReadyForServerCompletion: function(paymentId, txid) { return fetch('https://www.mydocassistant.online/api/complete', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ paymentId: paymentId, txid: txid }) }); }, onCancel: function(paymentId) { statusBox.innerText = "Payment cancelled."; payBtn.disabled = false; }, onError: function(error, payment) { statusBox.innerText = "Payment error: " + error.message; payBtn.disabled = false; } }); } // Step 2: App-to-User Payout async function handlePayout() { payoutBtn.disabled = true; statusBox.innerText = "Processing payout to your wallet..."; try { const response = await fetch('https://www.mydocassistant.online/api/payout', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ amount: "0.1", uid: currentUser.uid }) }); const data = await response.json(); if (response.ok) { statusBox.innerText = `Payout Successful!`; } else { statusBox.innerText = `Payout Failed: ${data.error || 'Unknown error'}`; } } catch (error) { statusBox.innerText = `Payout Error: ${error.message}`; } finally { payoutBtn.disabled = false; } }
